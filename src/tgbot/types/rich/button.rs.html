5/// Represents a style of the rich message button.
6#[derive(Clone, Copy, Debug, Deserialize, Serialize)]
7#[serde(rename_all = "snake_case")]
8pub enum RichMessageButtonStyle {
9    /// Danger.
10    Danger,
11    /// Success.
12    Success,
13    /// Primary.
14    Primary,
15    /// Link.
16    Link,
17}

19/// Represents a button in a rich message.
20///
21/// Exactly one of the fields other than text and style must be used to specify the type of the button.
22#[serde_with::skip_serializing_none]
23#[derive(Clone, Debug, Deserialize, Serialize)]
24pub struct RichMessageButton {
25    /// Text of the button.
26    pub text: RichText,
27    /// Type of the button.
28    #[serde(flatten)]
29    pub button_type: RichMessageButtonType,
30    /// Style of the button.
31    pub style: Option<RichMessageButtonStyle>,
32}
33
34impl<T> From<(T, RichMessageButtonType)> for RichMessageButton
35where
36    T: Into<RichText>,
37{
38    fn from((text, button_type): (T, RichMessageButtonType)) -> Self {
39        Self {
40            text: text.into(),
41            button_type,
42            style: None,
43        }
44    }
45}
46
47impl RichMessageButton {
48    /// Sets a new style for the button.
49    ///
50    /// # Arguments
51    ///
52    /// * `value` - The style to set.
53    pub fn with_style(mut self, value: RichMessageButtonStyle) -> Self {
54        self.style = Some(value);
55        self
56    }
57}

59/// Represents a rich message button type.
60#[derive(Clone, Debug, Deserialize, Serialize)]
61#[serde(rename_all = "snake_case")]
62pub enum RichMessageButtonType {
63    /// Data to be sent in a callaback query
64    /// to the bot when the button is pressed;
65    /// 1-64 bytes.
66    CallbackData(String),
67    /// A button that copies the specified text to the clipboard.
68    CopyText {
69        /// The text to be copied to the clipboard; 1-256 characters.
70        text: String,
71    },
72    /// The button is disabled and does nothing.
73    Disabled {},
74    /// An HTTPs URL used to automatically authorize the user.
75    ///
76    /// Can be used as a replacement for the Telegram Login Widget.
77    ///
78    /// Not supported for ephemeral messages.
79    LoginUrl(LoginUrl),
80    /// Pressing the button will prompt the user to select
81    /// one of their chats, open that chat and insert the bot's username
82    /// and the specified inline query in the input field.
83    ///
84    /// May be empty, in which case just the bot's username will be inserted.
85    ///
86    /// Not supported for messages sent in channel direct messages chats
87    /// and on behalf of a business account.
88    SwitchInlineQuery(String),
89    /// Pressing the button will insert the bot's username
90    /// and the specified inline query in the current chat's input field.
91    ///
92    /// May be empty, in which case only the bot's username will be inserted.
93    ///
94    /// Not supported in channels and for messages sent in
95    /// channel direct messages chats and on behalf of a business account.
96    SwitchInlineQueryCurrentChat(String),
97    /// Pressing the button will prompt the user
98    /// to select one of their chats of the specified type,
99    /// open that chat and insert the bot's username
100    /// and the specified inline query in the input field.
101    ///
102    /// Not supported for messages sent in channel direct messages chats
103    /// and on behalf of a business account.
104    SwitchInlineQueryChosenChat(SwitchInlineQueryChosenChat),
105    /// Pressing the button will insert the bot's username
106    /// and the specified inline query in the current chat's input field.
107    ///
108    /// May be empty, in which case only the bot's username will be inserted.
109    ///
110    /// Not supported in channels and for messages sent in
111    /// channel direct messages chats and on behalf of a business account.
112    Url(String),
113    /// Description of the Web App that will be
114    /// launched when the user presses the button.
115    ///
116    /// The Web App will be able to send an arbitrary message
117    /// on behalf of the user using the method AnswerWebAppQuery.
118    ///
119    /// Available only in private chats between a user and the bot.
120    ///
121    /// Not supported for the messages sent on behalf of a
122    /// business account.
123    WebApp(WebAppInfo),
124}
