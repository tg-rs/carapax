31/// Describes a rich message to be sent.
32#[derive(Debug)]
33pub struct InputRichMessage {
34    data: InputRichMessageData,
35    blocks: Option<Vec<InputRichBlock>>,
36    media: Option<Vec<(String, InputMedia)>>,
37}

39impl InputRichMessage {
40    fn new(data: InputRichMessageData) -> Self {
41        Self {
42            data,
43            blocks: None,
44            media: None,
45        }
46    }
47
48    /// Creates a new `InputRichMessage`.
49    ///
50    /// # Arguments
51    ///
52    /// * `value` - Content of the rich message to send described using Markdown formatting.
53    pub fn markdown<T>(value: T) -> Self
54    where
55        T: Into<String>,
56    {
57        Self::new(InputRichMessageData::markdown(value))
58    }
59
60    /// Creates a new `InputRichMessage`.
61    ///
62    /// # Arguments
63    ///
64    /// * `value` - Content of the rich message to send described using HTML formatting.
65    pub fn html<T>(value: T) -> Self
66    where
67        T: Into<String>,
68    {
69        Self::new(InputRichMessageData::html(value))
70    }
71
72    /// Creates a new `InputRichMessage`.
73    ///
74    /// # Arguments
75    ///
76    /// * `value` - Content of the rich message to send described as a list of blocks.
77    pub fn blocks<T>(value: T) -> Self
78    where
79        T: IntoIterator<Item = InputRichBlock>,
80    {
81        let mut result = Self::new(InputRichMessageData::default());
82        result.blocks = Some(value.into_iter().collect());
83        result
84    }
85
86    /// Sets a new value for the `is_rtl` flag.
87    ///
88    /// # Arguments
89    ///
90    /// * `value` - Whether the rich message must be shown right-to-left.
91    pub fn with_is_rtl(mut self, value: bool) -> Self {
92        self.data.is_rtl = Some(value);
93        self
94    }
95
96    /// Sets a new list of media.
97    ///
98    /// # Arguments
99    ///
100    /// * `value` - A sequence of `(id, media)` pairs.
101    ///
102    /// Media specified in the markdown or html fields using
103    /// `tg://(audio|photo|video)?id=` links.
104    pub fn with_media<A, B, C>(mut self, value: A) -> Self
105    where
106        A: IntoIterator<Item = (B, C)>,
107        B: Into<String>,
108        C: Into<InputMedia>,
109    {
110        self.media = Some(value.into_iter().map(|(id, media)| (id.into(), media.into())).collect());
111        self
112    }
113
114    /// Sets a new value for the `skip_entity_detection` flag.
115    ///
116    /// # Arguments
117    ///
118    /// * `value` - Whethert to skip automatic detection of entities in the text.
119    pub fn with_skip_entity_detection(mut self, value: bool) -> Self {
120        self.data.skip_entity_detection = Some(value);
121        self
122    }
123}

125impl WriteForm for InputRichMessage {
126    type Output = InputRichMessageData;
127
128    fn write(self, form: &mut Form) -> Self::Output {
129        let Self {
130            mut data,
131            blocks,
132            media,
133        } = self;
134        data.blocks = blocks.map(|items| items.into_iter().map(|x| x.write(form)).collect());
135        data.media = media.map(|items| {
136            items
137                .into_iter()
138                .map(|(id, media)| InputRichMessageMedia {
139                    id,
140                    media: media.write(form),
141                })
142                .collect()
143        });
144        data
145    }
146}

392/// Streams a partial rich message to a user while the message is being generated.
393///
394/// Note that streamed draft is ephemeral and acts as a temporary
395/// 30-second preview - once the output is finalized.
396///
397/// You must call [`crate::types::SendRichMessage`] with the complete message
398/// to persist it in the user's chat.
399#[derive(Debug)]
400pub struct SendRichMessageDraft {
401    chat_id: Integer,
402    draft_id: Integer,
403    rich_message: InputRichMessage,
404    can_stop: Option<bool>,
405    keep_on_stop: Option<bool>,
406    message_thread_id: Option<Integer>,
407}
408
409impl SendRichMessageDraft {
410    /// Creates a new `SendRichMessageDraft`.
411    ///
412    /// # Arguments
413    ///
414    /// * `chat_id` - Unique identifier of the target chat.
415    /// * `draft_id` - Unique identifier of the message draft; must be non-zero;
416    ///   changes to drafts with the same identifier are animated.
417    /// * `rich_message` - The partial message to be streamed.
418    pub fn new(chat_id: Integer, draft_id: Integer, rich_message: InputRichMessage) -> Self {
419        Self {
420            chat_id,
421            draft_id,
422            rich_message,
423            can_stop: None,
424            keep_on_stop: None,
425            message_thread_id: None,
426        }
427    }
428
429    /// Sets a new value for the `can_stop` flag.
430    ///
431    /// # Arguments
432    ///
433    /// * `value` - Whether to show the user a button to stop further drafts.
434    ///
435    /// The bot will receive an update "stopped_message_generation" if the user
436    /// presses the button.
437    pub fn with_can_stop(mut self, value: bool) -> Self {
438        self.can_stop = Some(value);
439        self
440    }
441
442    /// Sets a new value for the `keep_on_stop` flag.
443    ///
444    /// # Arguments
445    ///
446    /// * `value` - Whether to keep the draft in the chat when the button is pressed.
447    ///
448    /// The draft will still disappear after a short time or if the bot sends a message.
449    /// To fully preserve the partial draft, the bot should send it as a new message.
450    pub fn with_keep_on_stop(mut self, value: bool) -> Self {
451        self.keep_on_stop = Some(value);
452        self
453    }
454
455    /// Sets a new message thread ID.
456    ///
457    /// # Arguments
458    ///
459    /// * `value` - Unique identifier of the target message thread;
460    ///   for forum supergroups and private chats of bots with forum topic mode enabled only.
461    pub fn with_message_thread_id(mut self, value: Integer) -> Self {
462        self.message_thread_id = Some(value);
463        self
464    }
465}
