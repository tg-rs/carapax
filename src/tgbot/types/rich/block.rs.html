6/// Represents a block in a rich formatted message.
7#[derive(Debug, Clone, Deserialize, Serialize)]
8#[serde(from = "RawRichBlock", into = "RawRichBlock")]
9pub enum RichBlock {
10    /// An anchor (`<a>` with the `name` attribute).
11    Anchor(String),
12    /// An animation (`<video>`).
13    Animation(RichBlockAnimation),
14    /// An audio (`<audio>`).
15    Audio(RichBlockAudio),
16    /// A block quotation (`<blockquote>`).
17    BlockQuotation(RichBlockBlockQuotation),
18    /// A block containing a list of buttons that are show in one row,
19    /// corresponding to the custom HTML tag `<tg-button-row>`.
20    Buttons(RichBlockButtons),
21    /// A collage (`<tg-collage>`).
22    Collage(RichBlockCollage),
23    /// An expandable block for details disclosure (`<details>`).
24    Details(RichBlockDetails),
25    /// A divider (`<hr />`).
26    Divider,
27    /// A block with general file (`<tg-document>`).
28    Document(RichBlockDocument),
29    /// A block quotation with custom expandable attribute.
30    ExpandableBlockQuotation(RichBlockExpandableBlockQuotation),
31    /// A footer (`<footer>`).
32    Footer(RichText),
33    /// A list (`<ul>` or `<ol>`).
34    List(Vec<RichBlockListItem>),
35    /// A map (`<tg-map>`).
36    Map(RichBlockMap),
37    /// A mathematical expression (`<tg-math-block>`).
38    MathematicalExpression(String),
39    /// A paragraph (`<p>`).
40    Paragraph(RichText),
41    /// A Photo (`<photo>`).
42    Photo(RichBlockPhoto),
43    /// A preformatted text (`<pre>` or `<code>`).
44    Preformatted(RichBlockPreformatted),
45    /// A quotation with centered text (`<aside>`).
46    PullQuotation(RichBlockPullQuotation),
47    /// A section heading (`<h[1-6]>` text, size).
48    SectionHeading(RichText, Integer),
49    /// A slideshow (`<tg-slideshow>`).
50    Slideshow(RichBlockSlideshow),
51    /// A table (`<table>`).
52    Table(RichBlockTable),
53    /// A "Thinking ..." placeholder (`<tg-thinking>`).
54    ///
55    /// The block may be used only in `[crate::types::SendRichMessageDraft]`.
56    Thinking(RichText),
57    /// A video (`<video>`).
58    Video(RichBlockVideo),
59    /// A voice note (`<audio>`).
60    VoiceNote(RichBlockVoiceNote),
61}

473    Collage(RichBlockCollage),
474    Details(RichBlockDetails),
475    Divider,
476    Document(RichBlockDocument),
477    #[serde(rename = "expandable_blockquote")]
478    ExpandableBlockQuotation(RichBlockExpandableBlockQuotation),
479    Footer {
480        text: RichText,
481    },
482    List {
483        items: Vec<RichBlockListItem>,
484    },
485    Map(RichBlockMap),
486    MathematicalExpression {
487        expression: String,
488    },
489    Paragraph {
490        text: RichText,
491    },
492    Photo(RichBlockPhoto),
493    #[serde(rename = "pre")]
494    Preformatted(RichBlockPreformatted),
495    #[serde(rename = "pullquote")]
496    PullQuotation(RichBlockPullQuotation),
497    #[serde(rename = "heading")]
498    SectionHeading {
499        text: RichText,
500        size: Integer,
501    },
502    Slideshow(RichBlockSlideshow),
503    Table(RichBlockTable),
504    Thinking {
505        text: RichText,
506    },
507    Video(RichBlockVideo),
508    VoiceNote(RichBlockVoiceNote),
509}

511impl From<RichBlock> for RawRichBlock {
512    fn from(value: RichBlock) -> Self {
513        match value {
514            RichBlock::Anchor(name) => Self::Anchor { name },
515            RichBlock::Animation(value) => Self::Animation(value),
516            RichBlock::Audio(value) => Self::Audio(value),
517            RichBlock::BlockQuotation(value) => Self::BlockQuotation(value),
518            RichBlock::Buttons(value) => Self::Buttons(value),
519            RichBlock::Collage(value) => Self::Collage(value),
520            RichBlock::Details(value) => Self::Details(value),
521            RichBlock::Divider => Self::Divider,
522            RichBlock::Document(value) => Self::Document(value),
523            RichBlock::ExpandableBlockQuotation(value) => Self::ExpandableBlockQuotation(value),
524            RichBlock::Footer(text) => Self::Footer { text },
525            RichBlock::List(items) => Self::List { items },
526            RichBlock::Map(value) => Self::Map(value),
527            RichBlock::MathematicalExpression(expression) => Self::MathematicalExpression { expression },
528            RichBlock::Paragraph(text) => Self::Paragraph { text },
529            RichBlock::Photo(value) => Self::Photo(value),
530            RichBlock::Preformatted(value) => Self::Preformatted(value),
531            RichBlock::PullQuotation(value) => Self::PullQuotation(value),
532            RichBlock::SectionHeading(text, size) => Self::SectionHeading { text, size },
533            RichBlock::Slideshow(value) => Self::Slideshow(value),
534            RichBlock::Table(value) => Self::Table(value),
535            RichBlock::Thinking(text) => Self::Thinking { text },
536            RichBlock::Video(value) => Self::Video(value),
537            RichBlock::VoiceNote(value) => Self::VoiceNote(value),
538        }
539    }
540}
