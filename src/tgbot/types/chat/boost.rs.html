8/// Contains information about a chat boost.
9#[derive(Clone, Debug, Deserialize, PartialEq, PartialOrd, Serialize)]
10pub struct ChatBoost {
11    /// Point in time (Unix timestamp) when the chat was boosted.
12    pub add_date: Integer,
13    /// Unique identifier of the boost.
14    pub boost_id: String,
15    /// Point in time (Unix timestamp) when the boost will automatically expire,
16    /// unless the booster's Telegram Premium subscription is prolonged.
17    pub expiration_date: Integer,
18    /// Source of the added boost.
19    pub source: ChatBoostSource,
20}

35/// Describes the source of a chat boost.
36#[derive(Clone, Debug, Deserialize, PartialEq, PartialOrd, Serialize)]
37#[serde(rename_all = "snake_case", tag = "source")]
38pub enum ChatBoostSource {
39    /// The boost was obtained by the creation of Telegram Premium gift codes to boost a chat.
40    ///
41    /// Each such code boosts the chat 4 times for the duration of the corresponding Telegram Premium subscription.
42    /// Contains a user for which the gift code was created.
43    #[serde(
44        deserialize_with = "ChatBoostSourceUser::deserialize_value",
45        serialize_with = "ChatBoostSourceUser::serialize_value"
46    )]
47    GiftCode(User),
48    /// The boost was obtained by the creation of a Telegram Premium giveaway.
49    Giveaway(ChatBoostSourceGiveaway),
50    /// The boost was obtained by subscribing to Telegram Premium
51    /// or by gifting a Telegram Premium subscription to another user.
52    ///
53    /// Contains a user that boosted the chat
54    #[serde(
55        deserialize_with = "ChatBoostSourceUser::deserialize_value",
56        serialize_with = "ChatBoostSourceUser::serialize_value"
57    )]
58    Premium(User),
59}

61#[derive(Deserialize, Serialize)]
62struct ChatBoostSourceUser {
63    user: User,
64}
65
66impl ChatBoostSourceUser {
67    pub(super) fn deserialize_value<'de, D>(deserializer: D) -> Result<User, D::Error>
68    where
69        D: Deserializer<'de>,
70    {
71        Self::deserialize(deserializer).map(|x| x.user)
72    }
73
74    pub(super) fn serialize_value<S>(value: &User, serializer: S) -> Result<S::Ok, S::Error>
75    where
76        S: Serializer,
77    {
78        Self { user: value.clone() }.serialize(serializer)
79    }
80}

82/// The boost was obtained by the creation of a Telegram Premium giveaway.

86#[derive(Clone, Debug, Deserialize, PartialEq, PartialOrd, Serialize)]
87pub struct ChatBoostSourceGiveaway {

92    /// Whether the giveaway was completed, but there was no user to win the prize.

94    /// The number of Telegram Stars to be split between giveaway winners;
95    /// for Telegram Star giveaways only.
96    pub prize_star_count: Option<Integer>,
97    /// User that won the prize in the giveaway if any.
98    pub user: Option<User>,
99}

117/// Returns the list of boosts added to a chat by a user.
118///
119/// Requires administrator rights in the chat.
120#[derive(Clone, Debug, Serialize)]
121pub struct GetUserChatBoosts {
122    chat_id: ChatId,
123    user_id: Integer,
124}
125
126impl GetUserChatBoosts {
127    /// Creates a new `GetUserChatBoosts`.
128    ///
129    /// # Arguments
130    ///
131    /// * `chat_id` - Unique identifier for the chat.
132    /// * `user_id` - Unique identifier of the target user.
133    pub fn new<T>(chat_id: T, user_id: Integer) -> Self
134    where
135        T: Into<ChatId>,
136    {
137        Self {
138            chat_id: chat_id.into(),
139            user_id,
140        }
141    }
142}
