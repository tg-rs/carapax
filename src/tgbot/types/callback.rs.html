8/// Represents an incoming callback query from a callback button in an inline keyboard.
9///
10/// If the button that originated the query was attached to a message sent by the bot,
11/// the field message will be present.
12///
13/// If the button was attached to a message sent via the bot (in inline mode),
14/// the field `inline_message_id` will be present.
15///
16/// Exactly one of the fields data or `game_short_name` will be present.
17#[serde_with::skip_serializing_none]
18#[derive(Clone, Debug, Deserialize, Serialize)]
19pub struct CallbackQuery {
20    /// Sender of the query.
21    pub from: User,
22    /// Unique identifier of the query.
23    pub id: String,
24    /// Global identifier, uniquely corresponding
25    /// to the chat to which the message with the
26    /// callback button was sent.
27    ///
28    /// Useful for high scores in games.
29    pub chat_instance: Option<String>,
30    /// Data associated with the callback button.
31    ///
32    /// Be aware that a bad client can send arbitrary data in this field.
33    pub data: Option<String>,
34    /// Short name of a Game to be returned,
35    /// serves as the unique identifier for the game.
36    pub game_short_name: Option<String>,
37    /// Identifier of the message sent via the bot
38    /// in inline mode, that originated the query.
39    pub inline_message_id: Option<String>,
40    /// Message with the callback button that originated the query.
41    ///
42    /// Note that message content and message date
43    /// will not be available if the message is too old.
44    pub message: Option<MaybeInaccessibleMessage>,
45}

47/// Sends an answer to a callback query sent from an inline keyboard.
48///
49/// The answer will be displayed to the user as a notification at the top of the chat screen or as an alert.
50///
51/// Alternatively, the user can be redirected to the specified Game URL.
52///
53/// For this option to work, you must first create a game for your bot via Bot Father and accept the terms.
54///
55/// Otherwise, you may use links like `t.me/your_bot?start=XXX` that open your bot with a parameter.
56#[serde_with::skip_serializing_none]
57#[derive(Clone, Debug, Serialize)]
58pub struct AnswerCallbackQuery {
59    callback_query_id: String,
60    cache_time: Option<Integer>,
61    show_alert: Option<bool>,
62    text: Option<String>,
63    url: Option<String>,
64}
65
66impl AnswerCallbackQuery {
67    /// Creates a new `AnswerCallbackQuery`.
68    ///
69    /// # Arguments
70    ///
71    /// * `callback_query_id` - Unique identifier of the query to be answered.
72    pub fn new<T>(callback_query_id: T) -> Self
73    where
74        T: Into<String>,
75    {
76        Self {
77            callback_query_id: callback_query_id.into(),
78            cache_time: None,
79            show_alert: None,
80            text: None,
81            url: None,
82        }
83    }
84
85    /// Sets a new cache time.
86    ///
87    /// # Arguments
88    ///
89    /// * `value` - The maximum amount of time in seconds that the result
90    ///   of the callback query may be cached client-side;
91    ///   telegram apps will support caching starting in version 3.14;
92    ///   default - 0.
93    pub fn with_cache_time(mut self, value: Integer) -> Self {
94        self.cache_time = Some(value);
95        self
96    }
97
98    /// Sets a new value for the `short_alert` flag.
99    ///
100    /// # Arguments
101    ///
102    /// * `value` - An alert will be shown by the client instead
103    ///   of a notification at the top of the chat screen;
104    ///   default - `false`.
105    pub fn with_show_alert(mut self, value: bool) -> Self {
106        self.show_alert = Some(value);
107        self
108    }
109
110    /// Sets a new text.
111    ///
112    /// # Arguments
113    ///
114    /// * `value` - Text of the notification;
115    ///   if not specified, nothing will be shown to the user;
116    ///   0-200 characters.
117    pub fn with_text<T>(mut self, value: T) -> Self
118    where
119        T: Into<String>,
120    {
121        self.text = Some(value.into());
122        self
123    }
124
125    /// Sets a new URL.
126    ///
127    /// # Arguments
128    ///
129    /// * `value` - URL that will be opened by the user's client.
130    ///
131    /// If you have created a game and accepted the conditions via Bot Father,
132    /// specify the URL that opens your game – note that this will only work
133    /// if the query comes from a callback game button.
134    ///
135    /// Otherwise, you may use links like `t.me/your_bot?start=XXX`
136    /// that open your bot with a parameter.
137    pub fn with_url<T>(mut self, value: T) -> Self
138    where
139        T: Into<String>,
140    {
141        self.url = Some(value.into());
142        self
143    }
144}
