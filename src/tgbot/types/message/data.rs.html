600impl From<RawMessageData> for MessageData {
601    fn from(value: RawMessageData) -> Self {
602        match value {
603            RawMessageData::Animation(value) => Self::Animation(value),
604            RawMessageData::AutoDeleteTimerChanged(value) => Self::AutoDeleteTimerChanged(value),
605            RawMessageData::BoostAdded { boost_count } => Self::BoostAdded(boost_count),
606            RawMessageData::ChannelChatCreated(True) => Self::ChannelChatCreated,
607            RawMessageData::ChatBackgroundSet(value) => Self::ChatBackgroundSet(value),
608            RawMessageData::ChatOwnerChanged { new_owner } => Self::ChatOwnerChanged(new_owner),
609            RawMessageData::ChatOwnerLeft { new_owner } => Self::ChatOwnerLeft(new_owner),
610            RawMessageData::ChatShared(value) => Self::ChatShared(value),
611            RawMessageData::Checklist(value) => Self::Checklist(value),
612            RawMessageData::ChecklistTasksAdded(value) => Self::ChecklistTasksAdded(value),
613            RawMessageData::ChecklistTasksDone(value) => Self::ChecklistTasksDone(value),
614            RawMessageData::CommunityChatAdded(value) => Self::CommunityChatAdded(value),
615            RawMessageData::CommunityChatJoined(value) => Self::CommunityChatJoined(value),
616            RawMessageData::CommunityChatRemoved(value) => Self::CommunityChatRemoved(value),
617            RawMessageData::ConnectedWebsite(value) => Self::ConnectedWebsite(value),
618            RawMessageData::Contact(value) => Self::Contact(value),
619            RawMessageData::DeleteChatPhoto(True) => Self::DeleteChatPhoto,
620            RawMessageData::Dice(value) => Self::Dice(value),
621            RawMessageData::DirectMessagePriceChanged(value) => Self::DirectMessagePriceChanged(value),
622            RawMessageData::ForumTopicClosed {} => Self::ForumTopicClosed,
623            RawMessageData::ForumTopicCreated(value) => Self::ForumTopicCreated(value),
624            RawMessageData::ForumTopicEdited(value) => Self::ForumTopicEdited(value),
625            RawMessageData::ForumTopicReopened {} => Self::ForumTopicReopened,
626            RawMessageData::Game(value) => Self::Game(value),
627            RawMessageData::Gift(value) => Self::Gift(value),
628            RawMessageData::GiftUpgradeSent(value) => Self::GiftUpgradeSent(value),
629            RawMessageData::GeneralForumTopicHidden {} => Self::GeneralForumTopicHidden,
630            RawMessageData::GeneralForumTopicUnhidden {} => Self::GeneralForumTopicUnhidden,
631            RawMessageData::Giveaway(value) => Self::Giveaway(value),
632            RawMessageData::GiveawayCreated(value) => Self::GiveawayCreated(value),
633            RawMessageData::GiveawayCompleted(value) => Self::GiveawayCompleted(value),
634            RawMessageData::GiveawayWinners(value) => Self::GiveawayWinners(value),
635            RawMessageData::GroupChatCreated(True) => Self::GroupChatCreated,
636            RawMessageData::Invoice(value) => Self::Invoice(value),
637            RawMessageData::LeftChatMember(value) => Self::LeftChatMember(value),
638            RawMessageData::Location(value) => Self::Location(value),
639            RawMessageData::ManagedBotCreated(value) => Self::ManagedBotCreated(value),
640            RawMessageData::MigrateFromChatId(value) => Self::MigrateFromChatId(value),
641            RawMessageData::MigrateToChatId(value) => Self::MigrateToChatId(value),
642            RawMessageData::NewChatMembers(value) => Self::NewChatMembers(value),
