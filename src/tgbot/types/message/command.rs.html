7/// A simple command implementation.
8///
9/// We just take first command from a message and ignore others.
10/// Also we assume that entire text after command is arguments separated by whitespace.
11/// In order to include space in argument you need to wrap it with `'`: `'arg1 v' arg2`.
12///
13/// # Example
14/// ```
15/// use tgbot::types::{Command, Message};
16/// use std::convert::TryFrom;
17///
18/// fn handle_command(message: Message) {
19///     let command = Command::try_from(message).unwrap();
20///     println!("NAME: {}", command.get_name());
21///     println!("ARGUMENTS: {:?}", command.get_args());
22///     println!("MESSAGE: {:?}", command.get_message());
23/// }
24/// ```
25#[derive(Clone, Debug)]
26pub struct Command {
27    name: String,
28    args: Vec<String>,
29    message: Message,
30}
31
32impl Command {
33    /// Returns the name of the command with leading slash.
34    pub fn get_name(&self) -> &str {
35        &self.name
36    }
37
38    /// Returns the list of arguments.
39    pub fn get_args(&self) -> &[String] {
40        &self.args
41    }
42
43    /// Returns the message where the command comes from.
44    pub fn get_message(&self) -> &Message {
45        &self.message
46    }
47}
48
49/// Represents an error when parsing a command.
50#[derive(Debug)]
51pub enum CommandError {
52    /// A command is not found in a message.
53    NotFound,
54    /// Failed to create an UTF-16 string when reading a command from a message.
55    Utf16(FromUtf16Error),
56    /// An error when splitting an arguments string with mismatched quotes.
57    MismatchedQuotes(MismatchedQuotes),
58}
59
60impl From<FromUtf16Error> for CommandError {
61    fn from(err: FromUtf16Error) -> Self {
62        Self::Utf16(err)
63    }
64}
65
66impl From<MismatchedQuotes> for CommandError {
67    fn from(err: MismatchedQuotes) -> Self {
68        Self::MismatchedQuotes(err)
69    }
70}
71
72impl Error for CommandError {
73    fn source(&self) -> Option<&(dyn Error + 'static)> {
74        match self {
75            CommandError::NotFound => None,
76            CommandError::Utf16(err) => Some(err),
77            CommandError::MismatchedQuotes(_) => None,
78        }
79    }
80}
81
82impl fmt::Display for CommandError {
83    fn fmt(&self, out: &mut fmt::Formatter) -> fmt::Result {
84        write!(
85            out,
86            "failed to parse command: {}",
87            match self {
88                CommandError::NotFound => String::from("not found"),
89                CommandError::Utf16(err) => err.to_string(),
90                CommandError::MismatchedQuotes(_) => String::from("mismatched quotes"),
91            }
92        )
93    }
94}

96impl TryFrom<Message> for Command {
97    type Error = CommandError;
98
99    fn try_from(message: Message) -> Result<Self, Self::Error> {
100        match message.get_text().map(|text| (text.get_bot_commands(), text)) {
101            Some((Some(commands), text)) => {
102                // just take first command and ignore others
103                let command = &commands[0];
104                let name = command.command.clone();
105                // assume that all text after command is arguments
106                let offset = text.data.find(&name).unwrap_or(0);
107                // bot suffix is 1 character longer due to '@' symbol
108                let length = name.len() + command.bot_name.as_ref().map(|x| x.len() + 1).unwrap_or(0);
109                let pos = offset + length;
110                // pos is UTF-16 offset
111                let raw_args: Vec<u16> = text.data.encode_utf16().skip(pos).collect();
112                let raw_args = String::from_utf16(&raw_args)?;
113                let args = shellwords::split(&raw_args)?;
114                Ok(Command { name, args, message })
115            }
116            _ => Err(CommandError::NotFound),
117        }
118    }
119}

125    fn create_command(command: &str) -> Command {
126        let len = command.split_whitespace().next().unwrap().len();
127        let message: Message = serde_json::from_value(serde_json::json!(
128            {
129                "message_id": 1111,
130                "date": 0,
131                "from": {"id": 1, "is_bot": false, "first_name": "test"},
132                "chat": {"id": 1, "type": "private", "first_name": "test"},
133                "text": command,
134                "entities": [
135                    {"type": "bot_command", "offset": 0, "length": len}
136                ]
137            }
138        ))
139        .unwrap();
140        Command::try_from(message).unwrap()
141    }

144    fn command() {
