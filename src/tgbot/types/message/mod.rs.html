34/// Describes a message that can be inaccessible to the bot.
35#[derive(Clone, Debug, derive_more::From)]
36pub enum MaybeInaccessibleMessage {
37    /// Describes a message that was deleted or is otherwise inaccessible to the bot.
38    InaccessibleMessage(InaccessibleMessage),
39    /// Describes a regular message.
40    Message(Box<Message>),
41}
42
43impl<'de> Deserialize<'de> for MaybeInaccessibleMessage {
44    fn deserialize<D>(deserializer: D) -> Result<Self, D::Error>
45    where
46        D: Deserializer<'de>,
47    {
48        Message::deserialize(deserializer).map(|x| {
49            if x.date == 0 {
50                Self::InaccessibleMessage(InaccessibleMessage {
51                    chat: x.chat,
52                    message_id: x.id,
53                })
54            } else {
55                Self::Message(Box::new(x))
56            }
57        })
58    }
59}
60
61impl Serialize for MaybeInaccessibleMessage {
62    fn serialize<S>(&self, serializer: S) -> Result<S::Ok, S::Error>
63    where
64        S: serde::Serializer,
65    {
66        match self {
67            Self::InaccessibleMessage(value) => {
68                let value = RawInaccessibleMessage {
69                    message_id: value.message_id,
70                    chat: &value.chat,
71                    date: 0,
72                };
73                value.serialize(serializer)
74            }
75            Self::Message(value) => value.serialize(serializer),
76        }
77    }
78}

87/// Represents a message.
88#[serde_with::skip_serializing_none]
89#[derive(Clone, Debug, Deserialize, Serialize)]
90pub struct Message {
91    /// Chat the message belongs to.
92    pub chat: Chat,
93    /// Date the message was sent in Unix time.
94    pub date: Integer,
95    /// Date the message was last edited in Unix time.
96    pub edit_date: Option<Integer>,
97    /// Indicates whether the message can't be forwarded.
98    #[serde(default)]
99    pub has_protected_content: bool,
100    /// Unique message identifier inside the chat.
101    #[serde(rename = "message_id")]
102    pub id: Integer,
103    /// Indicates whether the message is a channel post that
104    /// was automatically forwarded
105    /// to the connected discussion group.
106    #[serde(default)]
107    pub is_automatic_forward: bool,
108    /// Sender of the message.
109    #[serde(flatten)]
110    pub sender: MessageSender,
111    /// Author signature.
112    pub author_signature: Option<String>,
113    /// Unique identifier of the business connection from which the message was received.
114    ///
115    /// If non-empty, the message belongs to a chat of the corresponding business account
116    /// that is independent from any potential bot chat which might share the same identifier.
117    pub business_connection_id: Option<String>,
118    /// Information about the direct messages chat topic that contains the message.
119    pub direct_messages_topic: Option<DirectMessagesTopic>,
120    /// Unique identifier of the message effect added to the message.
121    pub effect_id: Option<String>,
122    // For ephemeral messages, identifier of the ephemeral message inside this chat.
123    ///
124    /// The identifier may be reused for another ephemeral message
125    /// after the message is deleted or expires.
126    pub ephemeral_message_id: Option<Integer>,
127    /// Information about the message that is being replied to, which may come from another chat or forum topic.
128    pub external_reply: Option<ExternalReplyInfo>,
129    /// Information about the original message for forwarded messages.
130    pub forward_origin: Option<MessageOrigin>,
131    /// For a message sent by a guest bot, this is the information about the user and chat.
132    #[serde(flatten)]
133    pub guest_bot: Option<MessageGuestBot>,
134    /// The unique identifier for the guest query.
135    ///
136    /// Use this identifier with the method [`crate::types::AnswerGuestQuery`]
137    /// to send a response message.
138    ///
139    /// If non-empty, the message belongs to the chat where the guest bot was summoned,
140    /// which may not coincide with other existing bot chats sharing the same identifier.
141    pub guest_query_id: Option<String>,
142    /// Indicates whether the message media is covered by a spoiler animation.
143    pub has_media_spoiler: Option<bool>,
144    /// Whether the message was sent by an implicit action.
145    ///
146    /// For example, as an away or a greeting business message, or as a scheduled message.
147    pub is_from_offline: Option<bool>,
148    /// Whether the message is a paid post.
149    ///
150    /// Note that such posts must not be deleted for 24 hours to receive the payment and can't be edited.
151    pub is_paid_post: Option<bool>,
152    /// Indicates whether the message is sent to a topic in a forum supergroup or a private chat with the bot.
153    pub is_topic_message: Option<bool>,
154    /// Options used for link preview generation for the message,
155    /// if it is a text message and link preview options were changed.
156    pub link_preview_options: Option<LinkPreviewOptions>,
157    /// Unique identifier of a media message group this message belongs to.
158    pub media_group_id: Option<String>,
159    /// Unique identifier of a message thread to which the message belongs;
160    /// for supergroups and private chats only.
161    pub message_thread_id: Option<Integer>,
162    /// The number of Telegram Stars that were paid by the sender of the message to send it.
163    pub paid_star_count: Option<Integer>,
164    /// For replies that quote part of the original message, the quoted part of the message.
165    pub quote: Option<TextQuote>,
166    /// For ephemeral messages, the use who received the message.
167    pub receiver_user: Option<User>,
168    /// Inline keyboard attached to the message.
169    pub reply_markup: Option<InlineKeyboardMarkup>,
170    /// For replies, the original message or story.
171    #[serde(flatten)]
172    pub reply_to: Option<ReplyTo>,
173    /// Identifier of the specific checklist task that is being replied to.
174    pub reply_to_checklist_task_id: Option<Integer>,
175    /// Persistent identifier of the specific poll option that is being replied to.
176    pub reply_to_poll_option_id: Option<String>,
177    /// Number of boosts added by the user.
178    ///
179    /// Contains a value only if the sender of the message boosted the chat.
180    pub sender_boost_count: Option<Integer>,
181    /// The bot that actually sent the message on behalf of the business account.
182    ///
183    /// Available only for outgoing messages sent on behalf of the connected business account.
184    pub sender_business_bot: Option<User>,
185    /// Tag or custom title of the sender of the message; for supergroups only.
186    pub sender_tag: Option<String>,
187    /// Whether the caption must be shown above the message media.
188    pub show_caption_above_media: Option<bool>,
189    /// Information about suggested post parameters if the message is a suggested post in a channel direct messages chat.
190    ///
191    /// If the message is an approved or declined suggested post, then it can't be edited.
192    pub suggested_post_info: Option<SuggestedPostInfo>,
193    /// Bot through which the message was sent.
194    pub via_bot: Option<User>,
195
196    /// Contains message data.
197    #[serde(flatten)]
198    pub data: MessageData,
199}

201impl Message {
202    /// Returns `true` if the message has edited and `false` otherwise.
203    pub fn is_edited(&self) -> bool {
204        self.edit_date.is_some()
205    }
206
207    /// Returns a text of the message (includes caption).
208    pub fn get_text(&self) -> Option<&Text> {
209        match &self.data {
210            MessageData::Text(text) => Some(text),
211            MessageData::Audio(audio) => audio.caption.as_ref(),
212            MessageData::Document(doc) => doc.caption.as_ref(),
213            MessageData::Photo(photo) => photo.caption.as_ref(),
214            MessageData::Video(video) => video.caption.as_ref(),
215            MessageData::Voice(voice) => voice.caption.as_ref(),
216            _ => None,
217        }
218    }
219}
