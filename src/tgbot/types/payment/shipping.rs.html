8/// Represents a shipping address.
9#[derive(Clone, Debug, Deserialize, PartialEq, PartialOrd, Serialize)]
10pub struct ShippingAddress {
11    /// City.
12    pub city: String,
13    /// ISO 3166-1 alpha-2 country code.
14    pub country_code: String,
15    /// Address post code.
16    pub post_code: String,
17    /// State, if applicable.
18    pub state: String,
19    /// First line for the address.
20    pub street_line1: String,
21    /// Second line for the address.
22    pub street_line2: String,
23}

25/// Represents a shipping option.
26#[derive(Clone, Debug, Deserialize, PartialEq, PartialOrd, Serialize)]
27pub struct ShippingOption {
28    id: String,
29    title: String,
30    prices: Vec<LabeledPrice>,
31}
32
33impl ShippingOption {
34    /// Creates a new `ShippingOption`.
35    ///
36    /// # Arguments
37    ///
38    /// * `id` - Shipping option identifier.
39    /// * `title` - Option title.
40    /// * `prices` - List of price portions.
41    pub fn new<A, B, C>(id: A, title: B, prices: C) -> Self
42    where
43        A: Into<String>,
44        B: Into<String>,
45        C: IntoIterator<Item = LabeledPrice>,
46    {
47        Self {
48            id: id.into(),
49            title: title.into(),
50            prices: prices.into_iter().collect(),
51        }
52    }
53
54    /// Returns the ID of the option.
55    pub fn id(&self) -> &str {
56        &self.id
57    }
58
59    /// Returns the title of the option.
60    pub fn title(&self) -> &str {
61        &self.title
62    }
63
64    /// Returns the list of price portions.
65    pub fn prices(&self) -> &[LabeledPrice] {
66        &self.prices
67    }
68}

70/// Represents a shipping query.
71#[derive(Clone, Debug, Deserialize, PartialEq, PartialOrd, Serialize)]
72pub struct ShippingQuery {
73    /// Unique query identifier.
74    pub id: String,
75    /// User who sent the query.
76    pub from: User,
77    /// Bot specified invoice payload.
78    pub invoice_payload: String,
79    /// User specified shipping address.
80    pub shipping_address: ShippingAddress,
81}
82
83/// Replies to a shipping query.
84///
85/// If you sent an invoice requesting a shipping address
86/// and the parameter `is_flexible` was specified,
87/// the Bot API will send an [`crate::types::UpdateType::ShippingQuery`] to the bot.
88#[serde_with::skip_serializing_none]
89#[derive(Clone, Debug, Serialize)]
90pub struct AnswerShippingQuery {
91    ok: bool,
92    shipping_query_id: String,
93    error_message: Option<String>,
94    shipping_options: Option<Vec<ShippingOption>>,
95}
96
97impl AnswerShippingQuery {
98    /// Creates a new `AnswerShippingQuery` with a success answer.
99    ///
100    /// # Arguments
101    ///
102    /// * `id` - Unique identifier of the query to be answered.
103    /// * `options` - Array of available shipping options.
104    pub fn ok<A, B>(id: A, options: B) -> Self
105    where
106        A: Into<String>,
107        B: IntoIterator<Item = ShippingOption>,
108    {
109        Self {
110            ok: true,
111            shipping_query_id: id.into(),
112            error_message: None,
113            shipping_options: Some(options.into_iter().collect()),
114        }
115    }
116
117    /// Creates a new `AnswerShippingQuery` with an error answer.
118    ///
119    /// # Arguments
120    ///
121    /// * `id` - Unique identifier of the query to be answered.
122    /// * `message` - Error message in human readable form.
123    pub fn error<A, B>(id: A, message: B) -> Self
124    where
125        A: Into<String>,
126        B: Into<String>,
127    {
128        Self {
129            ok: false,
130            shipping_query_id: id.into(),
131            error_message: Some(message.into()),
132            shipping_options: None,
133        }
134    }
135}
136
137impl Method for AnswerShippingQuery {
138    type Response = bool;
139
140    fn into_payload(self) -> Result<Payload, PayloadError> {
141        Payload::json("answerShippingQuery", self)
142    }
143}
