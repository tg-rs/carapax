22/// Represents a content of a message to be sent as a result of an inline query.
23#[derive(Debug)]
24pub struct InputMessageContent {
25    data: InputMessageContentData,
26    input_rich_message: Option<InputRichMessage>,
27}
28
29impl InputMessageContent {
30    fn new(data: InputMessageContentData) -> Self {
31        Self {
32            data,
33            input_rich_message: None,
34        }
35    }
36
37    fn with_input_rich_message(mut self, value: InputRichMessage) -> Self {
38        self.input_rich_message = Some(value);
39        self
40    }
41}
42
43impl WriteForm for InputMessageContent {
44    type Output = InputMessageContentData;
45
46    fn write(self, form: &mut Form) -> Self::Output {
47        let Self {
48            mut data,
49            input_rich_message,
50        } = self;
51        data.rich_message = input_rich_message.map(|x| x.write(form));
52        data
53    }
54}

496    }

499impl<T> From<T> for InputMessageContentText
500where
501    T: Into<InputText>,
502{
503    fn from(value: T) -> Self {
504        let value = value.into();
505        let mut result = Self::new(value.data);
506        if let Some(entities) = value.entities {
507            result.data.entities = Some(entities);
508        } else if let Some(parse_mode) = value.parse_mode {
509            result.data.parse_mode = Some(parse_mode);
510        }
511        result
512    }
513}
